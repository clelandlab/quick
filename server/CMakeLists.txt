cmake_minimum_required(VERSION 3.16)
project(quick_server VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QUICK_STATIC_CXX_RUNTIME "Link the compiler C++ runtime statically for cross-built binaries" OFF)
if(QUICK_STATIC_CXX_RUNTIME)
  add_link_options(-static-libstdc++ -static-libgcc)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Build on Linux/PYNQ or with a Linux cross toolchain (Windows users can use WSL).")
endif()
find_package(Threads REQUIRED)
add_library(quick_core src/common.cpp src/wire.cpp src/server.cpp third_party/ryu/d2s.c)
target_include_directories(quick_core PUBLIC include third_party)
target_link_libraries(quick_core PUBLIC Threads::Threads)
target_compile_options(quick_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(quick_hardware src/metadata.cpp src/hardware.cpp third_party/pugixml.cpp)
target_link_libraries(quick_hardware PUBLIC quick_core ${CMAKE_DL_LIBS})
target_compile_options(quick_hardware PRIVATE -Wall -Wextra -Wpedantic)
add_executable(quick-server src/main.cpp)
target_link_libraries(quick-server PRIVATE quick_hardware)
install(TARGETS quick-server RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(quick-connection-tests tests/connections.cpp)
  target_link_libraries(quick-connection-tests PRIVATE quick_core)
  target_compile_options(quick-connection-tests PRIVATE -Wall -Wextra -Wpedantic)
  foreach(case fifo single_limit stalled_clients acquisition acquisition_takeover pending_takeover stop_at_limit)
    add_test(NAME connections_${case} COMMAND quick-connection-tests ${case})
    set_tests_properties(connections_${case} PROPERTIES TIMEOUT 15)
  endforeach()
endif()
